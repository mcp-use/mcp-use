Use a ready `MCPConnection` to list and call tools.

```typescript
import { MCPClient } from "@mcp-use/client";

const client = new MCPClient({
  mcpServers: { demo: { url: "http://localhost:3000/mcp" } },
});

try {
  const conn = await client.connect("demo");

  const tools = await conn.listTools();
  for (const tool of tools) {
    console.log(tool.name, tool.description);
  }

  const result = await conn.callTool("read_file", { path: "/tmp/a.txt" });

  if (result.isError) {
    console.error(result.content);
  } else {
    for (const block of result.content) {
      if (block.type === "text") console.log(block.text);
    }
  }
} finally {
  await client.close();
}
```

## Call an authenticated remote tool from Node.js

Use `authToken` to send a bearer token with requests to a remote Streamable
HTTP server. This example searches a handbook and reads the source URLs from
`structuredContent`. It uses [ClickHouse Loom](https://github.com/ClickHouse/loom).

Set `LOOM_MCP_URL` to your Loom server's HTTPS endpoint with
`/mcp?toolset=assistant`. Set `LOOM_NAMESPACE` to the indexed handbook's
namespace and `LOOM_VIEWER_TOKEN` to a viewer token scoped to that namespace.

```typescript
import { MCPClient } from "@mcp-use/client";

const url = process.env.LOOM_MCP_URL;
const authToken = process.env.LOOM_VIEWER_TOKEN;
const namespace = process.env.LOOM_NAMESPACE;
if (!url || !authToken || !namespace) {
  throw new Error("Set LOOM_MCP_URL, LOOM_VIEWER_TOKEN, and LOOM_NAMESPACE");
}

const client = new MCPClient({
  mcpServers: { loom: { url, authToken } },
});

try {
  const conn = await client.connect("loom");
  const result = await conn.callTool("loom_knowledge_search", {
    namespace,
    namespace_scope: "this",
    query: "When does the deployment freeze start?",
    search_mode: "lexical",
    rerank: "none",
    top_k: 3,
  });
  if (result.isError) {
    const detail = result.content.find((block) => block.type === "text");
    throw new Error(
      detail?.type === "text"
        ? `Loom document search failed: ${detail.text}`
        : "Loom document search failed",
    );
  }

  const data = result.structuredContent;
  if (
    !data ||
    typeof data !== "object" ||
    !("results" in data) ||
    !Array.isArray(data.results)
  ) {
    throw new Error("Expected structured search results");
  }
  for (const hit of data.results) {
    if (typeof hit?.source_uri === "string") console.log(hit.source_uri);
  }
} finally {
  await client.close();
}
```

See [Authentication](/typescript/client/authentication#bearer-tokens) for more
ways to configure credentials.

## Timeouts

Pass MCP request options as the third argument:

```typescript
const abortController = new AbortController();

await conn.callTool(
  "long_task",
  { input: "data" },
  {
    timeout: 300_000,
    resetTimeoutOnProgress: true,
    maxTotalTimeout: 900_000,
    signal: abortController.signal,
  },
);
```

## Lazy connect

```typescript
let conn = client.getSession("demo");
if (!conn) conn = await client.connect("demo");
await conn.callTool("echo", { message: "hello" });
```

`getSession` returns `null` when not connected. `requireSession` throws.
